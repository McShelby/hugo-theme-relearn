{{- $page := .page }} {{- $location := .location }} {{- partialCached "_relearn/dependencies.gotmpl" $page $page.Path }} {{- /* dependencies are loaded by ascending `weight` like everything else Hugo orders that way; it is optional and counts as zero if it is not given, and dependencies of equal weight keep the alphabetical order of their names. `theme` is pinned behind all of them whatever its weight says, because `theme.js` has to be the last of ours to run */}} {{- $deps := site.Params.relearn.dependencies }} {{- $ordered := slice }} {{- range $k, $v := $deps }} {{- if ne $k "theme" }} {{- /* the offset keeps a negative weight sortable, the padding keeps 10 from landing in front of 9, and the name breaks the tie */}} {{- $weight := add 1000000 (int (default 0 $v.weight)) }} {{- $ordered = $ordered | append (dict "key" $k "order" (printf "%010d-%s" $weight $k)) }} {{- end }} {{- end }} {{- $keys := slice }} {{- range sort $ordered "order" }} {{- $keys = $keys | append .key }} {{- end }} {{- if isset $deps "theme" }} {{- $keys = $keys | append "theme" }} {{- end }} {{- range $k := $keys }} {{- $v := index $deps $k }} {{- $needs := printf "needs%s" $v.name }} {{- $has := printf "has%s" $v.name }} {{- $hasnested := printf "relearnHasNested%s" $v.name }} {{- $wants := $page.Page.Store.Get $needs }} {{- if ne (printf "%T" $wants) "bool" }} {{- $wants = or ($page.Page.Store.Get $has) (and ($page.Page.Store.Get "relearnIsNested") ($page.Page.Store.Get $hasnested)) }} {{- end }} {{- if and $wants }} {{- if $v.location }} {{- warnf "DEPRECATED parameter 'location' for dependency '%s' configured in your hugo.toml, query the 'location' parameter inside your dependency loader instead; see https://mcshelby.github.io/hugo-theme-relearn/introduction/releasenotes/5/#5-27-0" $k }} {{- end }} {{- if or (not $v.location) (eq $location $v.location) }} {{- $dep := printf "dependencies/%s.html" $k }} {{- partial $dep (dict "page" $page "location" $location) }} {{- end }} {{- end }} {{- end }}
