    isHTML = true
    mediaType = 'text/html'
    name = 'print'
    path = '_print'
    permalink'ble = false
    outputFormats:
  print:
    baseName: index
    isHTML: true
    mediaType: text/html
    name: print
    path: _print
    permalink'ble: false
